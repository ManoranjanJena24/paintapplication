

// Get the canvas element from the DOM
const canvas = document.getElementById("canvas");
// Set the width of the canvas to the inner width of the window minus 60px
canvas.width = window.innerWidth - 60;
// Set the height of the canvas to 400px
canvas.height = 400;

// The code uses the Canvas API (getContext("2d")) to draw lines, shapes, and manage the drawing state. This API is only available on <canvas> elements.
let context = canvas.getContext("2d");

// Set the starting background color of the canvas to white
let start_background_color = "white";
// Fill the entire canvas with the starting background color (white)
context.fillStyle = start_background_color;

// Fill the canvas with the starting background color
context.fillRect(0, 0, canvas.width, canvas.height);

// Set the default drawing color to black
let draw_color = "black";
// Set the default line width for drawing to 2px
let draw_width = "2";
// Flag to track if the user is currently drawing
let is_drawing = false;
// Flag to track if the eraser is active
let is_eraser = false;

// Array to store canvas states for undo functionality
let restore_array = [];
// Index to keep track of the current state in the undo history
let index = -1;

let undoStack = JSON.parse(localStorage.getItem('undoStack'))||[];
let redoStack = JSON.parse(localStorage.getItem('redoStack'))||[];

// Function to change the drawing color when a user clicks on a color button
function change_color(element) {
    // Set the draw color to the background color of the clicked element
    draw_color = element.style.background;
}

// Add event listeners for touch and mouse events to handle drawing actions
canvas.addEventListener("touchstart", start, false); // Start drawing on touch start
canvas.addEventListener("touchmove", draw, false); // Draw while moving the touch
canvas.addEventListener("mousedown", start, false); // Start drawing on mouse down
canvas.addEventListener("mousemove", draw, false); // Draw while moving the mouse

// Add event listeners to stop drawing on touch and mouse events
canvas.addEventListener("touchend", stop, false); // Stop drawing on touch end
canvas.addEventListener("mouseup", stop, false); // Stop drawing on mouse up
canvas.addEventListener("mouseout", stop, false); // Stop drawing if mouse leaves the canvas

// Function to start the drawing when the mouse/touch begins
function start(event) {
    // Set the is_drawing flag to true (indicating drawing has started)
    is_drawing = true;
    // Begin a new path for drawing
    context.beginPath(); // Start a new path (so lines don't overlap previous ones)
    // Move the drawing position to where the user clicked or touched
    context.moveTo(event.clientX - canvas.offsetLeft, event.clientY - canvas.offsetTop);
    // Prevent default behavior for the event (e.g., prevent text selection)
    event.preventDefault();
}

// Function to toggle the eraser mode
function toggleErase() {
    is_eraser = !is_eraser;
    if (is_eraser) {
        draw_color = start_background_color;  // Set the color to the background color (for erasing)
        document.getElementById('eraseButton').textContent = 'Draw';  // Change button text to 'Draw'
    } else {
        draw_color = "black";  // Revert to the default drawing color
        document.getElementById('eraseButton').textContent = 'Erase';  // Change button text to 'Erase'
    }
}

// Function to handle the drawing while the mouse/touch moves
function draw(event) {
    if (is_drawing) {
        context.lineTo(event.clientX - canvas.offsetLeft, event.clientY - canvas.offsetTop);
        context.strokeStyle = draw_color;
        context.lineWidth = draw_width;
        context.lineCap = "round";
        context.lineJoin = "round";
        context.stroke();  
    }
    event.preventDefault();
}

// Function to stop drawing when the mouse/touch ends
function stop(event) {
    // If the user was drawing
    if (is_drawing) {
        // Finish the current stroke (line)
        context.stroke();
        // Close the current path to prevent unintended lines
        context.closePath();
        // Set the is_drawing flag to false (drawing stopped)
        is_drawing = false;
    }
    // Prevent default behavior
    event.preventDefault();

    // If the event type is not 'mouseout' (the mouse leaving the canvas)
    if (event.type != 'mouseout') {
        // Save the current canvas state to the restore array for undo functionality
        restore_array.push(context.getImageData(0, 0, canvas.width, canvas.height));
        // Increment the index (moving forward in the undo history)
        index += 1;
    }

    // Log the restore array (for debugging purposes)
    console.log(restore_array);
}

// Function to clear the canvas (e.g., when the user clicks the 'Clear' button)
function clear_canvas() {
    // Set the fill color to the starting background color
    context.fillStyle = start_background_color;
    // Clear the entire canvas area (remove everything)
    context.clearRect(0, 0, canvas.width, canvas.height);
    // Fill the canvas with the starting background color again (white)
    context.fillRect(0, 0, canvas.width, canvas.height);

    // Clear the restore array and reset the index (empty the undo history)
    restore_array = [];
    index = -1;
}

// Function to undo the last drawing action (e.g., when the user clicks the 'Undo' button)
function undo_last() {
    // If there are no actions in the restore array (index <= 0), clear the canvas
    if (index <= 0) {
        clear_canvas();
    } else {
        // Decrement the index (move back in the undo history)
        index -= 1;
        // Remove the last saved canvas state from the restore array
        restore_array.pop();
        // Restore the canvas to the previous state
        context.putImageData(restore_array[index], 0, 0);
    }
}

    downloadLink.download = 'canvas_image.png';

    downloadLink.href = canvas.toDataURL('image/png');
    
    console.log(downloadLink.href,"aa" );
    
    // Programmatically trigger a click on the anchor element to start the download
    downloadLink.click();
}

function saveToLocalStorage() {
    // Get the canvas data URL in PNG format
    let canvasData = canvas.toDataURL('image/png');
    
    // Save the canvas data URL to localStorage (using a key)
    localStorage.setItem('canvasImage', canvasData);
    
    alert("Canvas saved to localStorage.");
}



// Function to load the saved canvas state from localStorage
function loadFromLocalStorage() {
    // Get the saved canvas data URL from localStorage
    let savedCanvasData = localStorage.getItem('canvasImage');
    
    if (savedCanvasData) {
        // Create a new image element
        let img = new Image();
        
        // Set the image source to the saved canvas data URL
        img.src = savedCanvasData;
        
        // Once the image is loaded, draw it onto the canvas
        img.onload = function() {
            context.clearRect(0, 0, canvas.width, canvas.height);  // Clear the canvas first
            context.drawImage(img, 0, 0);  // Draw the image on the canvas
        };
        
        alert("Canvas loaded from localStorage.");
    } else {
        alert("No saved canvas found in localStorage.");
    }
}
